# Runtime dependencies for the API service.
#
# The domain layer (`answergap/`) is deliberately stdlib-only — it uses
# urllib.request rather than `requests`, so nothing here is needed to run the
# validation scripts. These three exist for `api/main.py` alone.
#
# Pinned to the versions this project is developed and tested against, so a
# Railway build cannot silently pick up a different FastAPI than the laptop.

fastapi==0.141.1
uvicorn==0.52.4
pydantic==2.13.4

# Postgres. `[binary]` ships the compiled libpq, so no system package and no
# build step is needed on Railway or on a laptop.
psycopg[binary]==3.3.4

# Connection pooling. Without it every database call pays a fresh TCP + TLS +
# auth round trip, which measured as seconds per request rather than
# milliseconds - see the performance note in CLAUDE.md.
psycopg-pool==3.3.1
